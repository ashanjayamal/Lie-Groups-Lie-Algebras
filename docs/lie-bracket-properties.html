Chapter 15 Lie Bracket Properties

set.seed(321)
X <- matrix(rnorm(9), 3, 3)
Y <- matrix(rnorm(9), 3, 3)
Z <- matrix(rnorm(9), 3, 3)

jacobi_sum <- commutator(X, commutator(Y, Z)) +
              commutator(Y, commutator(Z, X)) +
              commutator(Z, commutator(X, Y))

bracket_props <- data.frame(
  Property = c("Antisymmetry: [X,Y] = -[Y,X]", 
               "Jacobi Identity",
               "so(3) closure"),
  Verified = c(
    max(abs(commutator(X, Y) + commutator(Y, X))) < 1e-13,
    max(abs(jacobi_sum)) < 1e-12,
    max(abs(commutator(skew_symmetric(rnorm(3)), 
                      skew_symmetric(rnorm(3))) + 
           t(commutator(skew_symmetric(rnorm(3)), 
                       skew_symmetric(rnorm(3)))))) < 1e-13
  )
)

kable(bracket_props, caption = "Lie Bracket Properties") %>%
  kable_styling() %>%
  column_spec(2, bold = TRUE, color = ifelse(bracket_props$Verified, "green", "red"))
