Chapter 9 Matrix Exponential Algorithms

matrix_exp_series <- function(X, terms = 20) {
  n <- nrow(X)
  result <- diag(n)
  term <- diag(n)
  for (m in 1:terms) {
    term <- (term %*% X) / m
    result <- result + term
  }
  return(result)
}

matrix_exp_eigen <- function(X) {
  eigen_decomp <- eigen(X)
  V <- eigen_decomp$vectors
  D <- diag(exp(eigen_decomp$values))
  return(Re(V %*% D %*% solve(V)))
}
set.seed(12345)
X_test <- matrix(rnorm(16), 4, 4)

t1 <- system.time({exp1 <- matrix_exp_series(X_test)})
t2 <- system.time({exp2 <- matrix_exp_eigen(X_test)})
t3 <- system.time({exp3 <- expm(X_test)})

results <- data.frame(
  Method = c("Power Series", "Eigenvalue", "expm (Padé)"),
  Time_sec = c(t1[3], t2[3], t3[3]),
  Error = c(max(abs(exp1 - exp3)), max(abs(exp2 - exp3)), 0)
)

kable(results, digits = 6, caption = "Algorithm Comparison") %>%
  kable_styling(bootstrap_options = c("striped", "hover"))
